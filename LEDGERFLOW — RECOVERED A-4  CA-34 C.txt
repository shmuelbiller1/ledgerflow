import React, { useMemo } from 'react';
import { AlertTriangle, CheckCircle2, Scale } from 'lucide-react';
import { useLedgerStore } from '../../stores/useLedgerStore';
import { useUIStore } from '../../stores/useUIStore';
import { useAuthStore } from '../../stores/useAuthStore';

/**
 * Since when don't the books balance? (A-4, CA-34): the panel on Data Health that answers it. When
 * the balance sheet is off, `src/lib/balanceBreak.ts` halves the date range of the entries to find
 * the first day it was off, and this shows that day's entries with the amount each contributes,
 * each opening in the usual record viewer.
 *
 * Read-only, like the rest of Data Health: it says what it found and changes nothing. The figures
 * come from the report code (`computeBalanceSheet`) over the history loaded here, the same entries
 * the checks above use.
 */
export default function BalanceBreakPanel() {
  const transactions = useLedgerStore(s => s.transactions);
  const accounts = useLedgerStore(s => s.accounts);
  const currency = useAuthStore(s => s.getCurrentWorkspace()?.currency) || 'USD';
  const openRecord = useUIStore(s => s.openRecord);
  const today = getLocalDateISO();
  const result = useMemo(() => findBalanceBreak(accounts || [], transactions || [], today), [accounts, transactions, today]);
  const money = (cents: number) => formatCurrency(cents, currency);
  return (
    <section className="rounded-2xl border border-[var(--line)] bg-[var(--surface)] p-5 space-y-3" aria-labelledby="balance-break">
      <h2 id="balance-break" className="text-sm font-bold flex items-center gap-2 text-[var(--text)]">
        <Scale size={16} className="text-[var(--brass)]" aria-hidden="true" />
        Since when don&apos;t the books balance?
      </h2>
      {result.balanced ? (
        <p className="text-sm text-[var(--pos)] flex items-center gap-2">
          <CheckCircle2 size={16} aria-hidden="true" />
          The books balance on {result.endDate}: what the workspace owns and what is owed against it agree to the cent.
        </p>
      ) : (
        <>
          <p className="text-sm text-amber-700 dark:text-amber-400 flex items-start gap-2">
            <AlertTriangle size={16} className="mt-0.5 shrink-0" aria-hidden="true" />
            <span>
              The books stopped balancing on <strong>{result.firstUnbalancedDate}</strong>
              {result.lastBalancedDate
                ? <>; they still balanced on {result.lastBalancedDate}</>
                : <> — the first entries in the loaded history are already off</>}
              . Found in {result.probes} report probe{result.probes === 1 ? '' : 's'}, by halving the dates that carry an entry.
            </span>
          </p>
          <p className="text-xs text-[var(--muted)]">
            At {result.endDate} the balance sheet is off by {money(Math.abs(result.differenceCents))}
            {result.differenceCents > 0
              ? ' — it shows more assets than the liabilities, equity and profit account for.'
              : ' — it shows more owed against the assets than the assets hold.'}
            {result.differenceBeforeCents === 0
              ? ' Nothing before that day moved the difference.'
              : ` It was already off by ${money(Math.abs(result.differenceBeforeCents))} before that day.`}
          </p>
          <ul className="space-y-2">
            {result.entries.map(entry => (
              <li key={entry.transactionId} className="p-4 rounded-xl border border-[var(--line)] flex flex-col sm:flex-row sm:items-center justify-between gap-3">
                <div className="text-[var(--text)]">
                  <p className="text-sm font-semibold">{entry.description}</p>
                  <p className="text-xs text-[var(--muted)] mt-0.5">
                    {entry.date} · {entry.offCents === 0
                      ? 'its own lines balance; it moves the difference through a line the sheet cannot place'
                      : `off by ${money(Math.abs(entry.offCents))}${entry.offCents > 0 ? ' (adds to the difference)' : ' (takes away from it)'}`}
                    {entry.missingAccounts.length > 0 && ` · it uses ${entry.missingAccounts.join(', ')}, which ${entry.missingAccounts.length === 1 ? 'is' : 'are'} not in the chart of accounts`}
                  </p>
                </div>
                <button
                  type="button"
                  onClick={() => openRecord({ type: 'transaction', id: entry.transactionId })}
                  className="px-3 py-1.5 text-xs font-semibold rounded-lg border border-[var(--line)] text-[var(--text)] shrink-0 self-start sm:self-auto"
                >
                  Open the entry
                </button>
              </li>
            ))}
            {result.entries.length === 0 && (
              <li className="text-xs text-[var(--muted)]">
                No entry is dated {result.firstUnbalancedDate} in the loaded history, so an opening balance or an entry loaded from another day is the cause.
              </li>
            )}
          </ul>
          <p className="text-xs text-[var(--muted)]">
            Read-only: this check never changes anything. Open the entry and correct its lines, or reverse it and enter it again.
          </p>
        </>
      )}
    </section>
  );
}

/**
 * Since when don't the books balance? (A-4, CA-34, AI #3)
 *
 * When the balance sheet doesn't balance, the answer used to be "somewhere in your history". This
 * finds the day it started by halving the range of dates that have entries: a balance is the same
 * on a day with no entries as on the day before, so only days that carry an entry (and the end
 * date) can be where it broke. Every probe runs the report code's own `computeBalanceSheet` — this
 * module computes no money of its own and keeps no second copy of the report rules — and the first
 * date whose figures are off is the answer.
 *
 * Then the entries of that day are added one at a time, and each entry's amount is what it changed
 * the difference by when it was added — so the numbers add up to that day's change, and the day's
 * biggest share is listed first. An entry can be off by zero and still be the cause: a line whose
 * account is not in the chart is left out of the sheet, and that shows as the amount the entry
 * takes out (and the missing account is named).
 *
 * Read-only: nothing here posts, repairs or changes anything.
 */
import type { ChartAccount, Transaction } from '../types';
import { computeBalanceSheet } from '../server/reports/computeReports';
import { isPostable, transactionDate } from './txFilters';
import { getLocalDateISO } from './utils';
export interface BalanceBreakEntry {
  transactionId: string;
  date: string;
  description: string;
  /** what this entry does to the difference: its own share of the imbalance, in cents */
  offCents: number;
  /** lines whose account is not in the chart of accounts (the sheet leaves them out) */
  missingAccounts: string[];
}
export interface BalanceBreakResult {
  /** whether the books balance at the end date */
  balanced: boolean;
  /** the date the search ran to (the workspace's today) */
  endDate: string;
  /** how far off the books are at the end date, in cents (0 when they balance) */
  differenceCents: number;
  /** the first date the books did not balance, or null when they always did */
  firstUnbalancedDate: string | null;
  /** the last date they still balanced before that, or null when no earlier entry exists */
  lastBalancedDate: string | null;
  /** how far off they already were before the break day (0 when it broke on the first day) */
  differenceBeforeCents: number;
  /** that day's entries, the biggest share first */
  entries: BalanceBreakEntry[];
  /** how many report probes the search used, for the sentence */
  probes: number;
}
/** The dates worth probing: every day a postable entry falls on, in order, then the end date. */
export function probeDates(transactions: Transaction[], endDate: string): string[] {
  const dates = new Set<string>();
  for (const tx of transactions) {
    if (!isPostable(tx)) continue;
    const date = transactionDate(tx);
    if (date && date <= endDate) dates.add(date);
  }
  dates.add(endDate);
  return [...dates].sort();
}
/** How far the balance sheet is off on a date, from the report code's own two totals. */
export function balanceDifferenceCents(accounts: ChartAccount[], transactions: Transaction[], endDate: string): number {
  const sheet = computeBalanceSheet(accounts, transactions, endDate);
  return sheet.totalAssets - sheet.totalClaims;
}
/**
 * The day the books stopped balancing, and what each entry of that day is off by. `endDate`
 * defaults to today; the search never looks past it.
 */
export function findBalanceBreak(accounts: ChartAccount[], transactions: Transaction[], endDate = getLocalDateISO()): BalanceBreakResult {
  const empty: BalanceBreakResult = {
    balanced: true, endDate, differenceCents: 0, firstUnbalancedDate: null, lastBalancedDate: null,
    differenceBeforeCents: 0, entries: [], probes: 0,
  };
  const dates = probeDates(transactions, endDate);
  if (dates.length === 0) return empty;
  let probes = 0;
  const balancedOn = (date: string): boolean => {
    probes += 1;
    const sheet = computeBalanceSheet(accounts, transactions, date);
    return sheet.isBalanced;
  };
  const last = dates[dates.length - 1];
  if (balancedOn(last)) return { ...empty, probes };
  // the last date is known to be off, so the first off date is at or before it: halve the range
  let low = 0;
  let high = dates.length - 1;
  while (low < high) {
    const middle = (low + high) >> 1;
    if (balancedOn(dates[middle])) low = middle + 1;
    else high = middle;
  }
  const firstUnbalancedDate = dates[low];
  const lastBalancedDate = low > 0 ? dates[low - 1] : null;
  const differenceCents = balanceDifferenceCents(accounts, transactions, last);
  // the entries of the break day, each with its own share of the difference (in the order they
  // come; the shares are per entry, so which one "gets" it does not depend on the others)
  const postable = transactions.filter(isPostable);
  const before = postable.filter(tx => transactionDate(tx) < firstUnbalancedDate);
  const thatDay = postable.filter(tx => transactionDate(tx) === firstUnbalancedDate);
  const missing = new Map<string, string[]>();
  for (const line of computeBalanceSheet(accounts, thatDay, firstUnbalancedDate).orphanedLines) {
    if (!line.transactionId) continue;
    missing.set(line.transactionId, [...(missing.get(line.transactionId) || []), line.accountId]);
  }
  const differenceBeforeCents = balanceDifferenceCents(accounts, before, firstUnbalancedDate);
  // add the day's entries one at a time: each entry's amount is what it changed the difference by
  const added = [...before];
  const entries: BalanceBreakEntry[] = thatDay.map(tx => {
    const differenceBeforeIt = balanceDifferenceCents(accounts, added, firstUnbalancedDate);
    added.push(tx);
    const after = balanceDifferenceCents(accounts, added, firstUnbalancedDate);
    const offCents = after - differenceBeforeIt;
    return {
      transactionId: tx.id,
      date: transactionDate(tx),
      description: tx.description || tx.id,
      offCents,
      missingAccounts: missing.get(tx.id) || [],
    };
  });
  entries.sort((a, b) => Math.abs(b.offCents) - Math.abs(a.offCents) || a.transactionId.localeCompare(b.transactionId));
  return { balanced: false, endDate: last, differenceCents, firstUnbalancedDate, lastBalancedDate, differenceBeforeCents, entries, probes };
}

/**
 * Since when don't the books balance? (A-4, CA-34): the search behind the Data Health panel.
 *
 * Every number here is worked out by hand from the report code's own rule (assets against
 * liabilities + equity + profit, whole cents):
 *   a sale of 1000 (bank +1000, sales −1000) adds 1000 to the assets and 1000 to profit → still level;
 *   an entry with both sides on one account, or a side missing, leaves the sheet off by the amount
 *   it does not account for. The dates are chosen so the halving has to look at the first day, the
 *   last day, a day in the middle, and a history that never breaks.
 */
import test, { describe } from 'node:test';
import assert from 'node:assert/strict';
import { balanceDifferenceCents, findBalanceBreak, probeDates } from '../lib/balanceBreak';
import { makeChartAccount, makeTransaction } from './fixtures';
import type { ChartAccount, Transaction } from '../types';
const bank: ChartAccount = makeChartAccount({ id: 'bank', code: '1010', name: 'Bank', type: 'asset', category: 'Cash' });
const sales: ChartAccount = makeChartAccount({ id: 'sales', code: '4000', name: 'Sales', type: 'income', category: 'Revenue' });
const rent: ChartAccount = makeChartAccount({ id: 'rent', code: '6100', name: 'Rent', type: 'expense', category: 'Operating Expenses' });
const loan: ChartAccount = makeChartAccount({ id: 'loan', code: '2500', name: 'Bank loan', type: 'liability', category: 'Long-term Liabilities' });
const accounts = [bank, sales, rent, loan];
const entry = (id: string, date: string, lines: Array<[string, number]>) => makeTransaction({
  id, date, description: `Entry ${id}`,
  lines: lines.map(([accountId, amountCents]) => ({ id: `line-${id}-${accountId}`, accountId, amountCents })),
});
/** A sale: the bank up, sales up; balanced. */
const sale = (id: string, date: string, cents: number) => entry(id, date, [['bank', cents], ['sales', -cents]]);
const today = '2026-03-31';
describe('Probing dates', () => {
  test('only days that carry a postable entry, in order, then the end date', () => {
    const transactions: Transaction[] = [
      sale('c', '2026-03-20', 100),
      sale('a', '2026-03-01', 100),
      makeTransaction({ id: 'draft', date: '2026-03-05', status: 'draft', lines: [{ accountId: 'bank', amountCents: 1 }] }),
      makeTransaction({ id: 'void', date: '2026-03-06', status: 'void', lines: [{ accountId: 'bank', amountCents: 1 }] }),
      sale('late', '2026-04-02', 100),
      makeTransaction({ id: 'stamp', date: '2026-03-10T12:00:00.000Z', lines: [{ accountId: 'bank', amountCents: 1 }, { accountId: 'sales', amountCents: -1 }] }),
    ];
    assert.deepEqual(probeDates(transactions, today), ['2026-03-01', '2026-03-10', '2026-03-20', today]);
  });
  test('the difference is the report code’s own two totals', () => {
    assert.equal(balanceDifferenceCents(accounts, [sale('a', '2026-03-01', 1000)], today), 0);
    assert.equal(balanceDifferenceCents(accounts, [entry('b', '2026-03-01', [['bank', 500], ['sales', -400]])], today), 100);
    assert.equal(balanceDifferenceCents(accounts, [], today), 0);
  });
});
describe('Finding the day the books stopped balancing', () => {
  test('a break on the first day of the history', () => {
    const transactions = [entry('first', '2026-03-01', [['bank', 500], ['sales', -400]])];
    const result = findBalanceBreak(accounts, transactions, today);
    assert.equal(result.balanced, false);
    assert.equal(result.firstUnbalancedDate, '2026-03-01');
    assert.equal(result.lastBalancedDate, null, 'nothing came before it');
    assert.equal(result.differenceCents, 100);
    assert.equal(result.differenceBeforeCents, 0);
    assert.deepEqual(result.entries.map(e => [e.transactionId, e.offCents]), [['first', 100]]);
    assert.equal(result.probes, 2, 'the end date and the first day');
  });
  test('a break on the last day', () => {
    const transactions = [
      sale('jan', '2026-03-01', 1000),
      entry('rent', '2026-03-05', [['rent', 300], ['bank', -200]]),
    ];
    const result = findBalanceBreak(accounts, transactions, today);
    assert.equal(result.firstUnbalancedDate, '2026-03-05');
    assert.equal(result.lastBalancedDate, '2026-03-01', 'the day before still balanced');
    assert.equal(result.differenceCents, 100, '800 in the bank against 700 of profit');
    assert.deepEqual(result.entries.map(e => [e.transactionId, e.offCents]), [['rent', 100]]);
    assert.equal(result.probes, 3);
  });
  test('a break in the middle, with balanced entries after it', () => {
    const transactions = [
      sale('a', '2026-03-01', 1000),
      entry('b', '2026-03-10', [['rent', 50], ['bank', -40]]),
      sale('c', '2026-03-20', 500),
    ];
    const result = findBalanceBreak(accounts, transactions, today);
    assert.equal(result.firstUnbalancedDate, '2026-03-10');

+/**
+ * Since when don't the books balance? (A-4, CA-34, AI #3)
+ *
+ * When the balance sheet doesn't balance, the answer used to be "somewhere in your history". This
+ * finds the day it started by halving the range of dates that have entries: a balance is the same
+ * on a day with no entries as on the day before, so only days that carry an entry (and the end
+ * date) can be where it broke. Every probe runs the report code's own `computeBalanceSheet` — this
+ * module computes no money of its own and keeps no second copy of the report rules — and the first
+ * date whose figures are off is the answer.
+ *
+ * Then the entries of that day are added one at a time, and each entry's amount is what it changed
+ * the difference by when it was added — so the numbers add up to that day's change, and the day's
+ * biggest share is listed first. An entry can be off by zero and still be the cause: a line whose
+ * account is not in the chart is left out of the sheet, and that shows as the amount the entry
+ * takes out (and the missing account is named).
+ *
+ * Read-only: nothing here posts, repairs or changes anything.
+ */
+import type { ChartAccount, Transaction } from '../types';
+import { computeBalanceSheet } from '../server/reports/computeReports';
+import { isPostable, transactionDate } from './txFilters';
+import { getLocalDateISO } from './utils';
+
+export interface BalanceBreakEntry {
+  transactionId: string;
+  date: string;
+  description: string;
+  /** what this entry does to the difference: its own share of the imbalance, in cents */
+  offCents: number;
+  /** lines whose account is not in the chart of accounts (the sheet leaves them out) */
+  missingAccounts: string[];
+}
+
+export interface BalanceBreakResult {
+  /** whether the books balance at the end date */
+  balanced: boolean;
+  /** the date the search ran to (the workspace's today) */
+  endDate: string;
+  /** how far off the books are at the end date, in cents (0 when they balance) */
+  differenceCents: number;
+  /** the first date the books did not balance, or null when they always did */
+  firstUnbalancedDate: string | null;
+  /** the last date they still balanced before that, or null when no earlier entry exists */
+  lastBalancedDate: string | null;
+  /** how far off they already were before the break day (0 when it broke on the first day) */
+  differenceBeforeCents: number;
+  /** that day's entries, the biggest share first */
+  entries: BalanceBreakEntry[];
+  /** how many report probes the search used, for the sentence */
+  probes: number;
+}
+
+/** The dates worth probing: every day a postable entry falls on, in order, then the end date. */
+export function probeDates(transactions: Transaction[], endDate: string): string[] {
+  const dates = new Set<string>();
+  for (const tx of transactions) {
+    if (!isPostable(tx)) continue;
+    const date = transactionDate(tx);
+    if (date && date <= endDate) dates.add(date);
+  }
+  dates.add(endDate);
+  return [...dates].sort();
+}
+
+/** How far the balance sheet is off on a date, from the report code's own two totals. */
+export function balanceDifferenceCents(accounts: ChartAccount[], transactions: Transaction[], endDate: string): number {
+  const sheet = computeBalanceSheet(accounts, transactions, endDate);
+  return sheet.totalAssets - sheet.totalClaims;
+}
+
+/**
+ * The day the books stopped balancing, and what each entry of that day is off by. `endDate`
+ * defaults to today; the search never looks past it.
+ */
+export function findBalanceBreak(accounts: ChartAccount[], transactions: Transaction[], endDate = getLocalDateISO()): BalanceBreakResult {
+  const empty: BalanceBreakResult = {
+    balanced: true, endDate, differenceCents: 0, firstUnbalancedDate: null, lastBalancedDate: null,
+    differenceBeforeCents: 0, entries: [], probes: 0,
+  };
+  const dates = probeDates(transactions, endDate);
+  if (dates.length === 0) return empty;
+
+  let probes = 0;
+  const balancedOn = (date: string): boolean => {
+    probes += 1;
+    const sheet = computeBalanceSheet(accounts, transactions, date);
+    return sheet.isBalanced;
+  };
+
+  const last = dates[dates.length - 1];
+  if (balancedOn(last)) return { ...empty, probes };
+
+  // the last date is known to be off, so the first off date is at or before it: halve the range
+  let low = 0;
+  let high = dates.length - 1;
+  while (low < high) {
+    const middle = (low + high) >> 1;
+    if (balancedOn(dates[middle])) low = middle + 1;
+    else high = middle;
+  }
+
+  const firstUnbalancedDate = dates[low];
+  const lastBalancedDate = low > 0 ? dates[low - 1] : null;
+  const differenceCents = balanceDifferenceCents(accounts, transactions, last);
+
+  // the entries of the break day, each with its own share of the difference (in the order they
+  // come; the shares are per entry, so which one "gets" it does not depend on the others)
+  const postable = transactions.filter(isPostable);
+  const before = postable.filter(tx => transactionDate(tx) < firstUnbalancedDate);
+  const thatDay = postable.filter(tx => transactionDate(tx) === firstUnbalancedDate);
+  const missing = new Map<string, string[]>();
+  for (const line of computeBalanceSheet(accounts, thatDay, firstUnbalancedDate).orphanedLines) {
+    if (!line.transactionId) continue;
+    missing.set(line.transactionId, [...(missing.get(line.transactionId) || []), line.accountId]);
+  }
+
+  const differenceBeforeCents = balanceDifferenceCents(accounts, before, firstUnbalancedDate);
+  // add the day's entries one at a time: each entry's amount is what it changed the difference by
+  const added = [...before];
+  const entries: BalanceBreakEntry[] = thatDay.map(tx => {
+    const differenceBeforeIt = balanceDifferenceCents(accounts, added, firstUnbalancedDate);
+    added.push(tx);
+    const after = balanceDifferenceCents(accounts, added, firstUnbalancedDate);
+    const offCents = after - differenceBeforeIt;
+    return {
+      transactionId: tx.id,
+      date: transactionDate(tx),
+      description: tx.description || tx.id,
+      offCents,
+      missingAccounts: missing.get(tx.id) || [],
+    };
+  });
+  entries.sort((a, b) => Math.abs(b.offCents) - Math.abs(a.offCents) || a.transactionId.localeCompare(b.transactionId));
+
+  return { balanced: false, endDate: last, differenceCents, firstUnbalancedDate, lastBalancedDate, differenceBeforeCents, entries, probes };
+}

+/**
+ * Since when don't the books balance? (A-4, CA-34): the search behind the Data Health panel.
+ *
+ * Every number here is worked out by hand from the report code's own rule (assets against
+ * liabilities + equity + profit, whole cents):
+ *   a sale of 1000 (bank +1000, sales −1000) adds 1000 to the assets and 1000 to profit → still level;
+ *   an entry with both sides on one account, or a side missing, leaves the sheet off by the amount
+ *   it does not account for. The dates are chosen so the halving has to look at the first day, the
+ *   last day, a day in the middle, and a history that never breaks.
+ */
+import test, { describe } from 'node:test';
+import assert from 'node:assert/strict';
+import { balanceDifferenceCents, findBalanceBreak, probeDates } from '../lib/balanceBreak';
+import { makeChartAccount, makeTransaction } from './fixtures';
+import type { ChartAccount, Transaction } from '../types';
+
+const bank: ChartAccount = makeChartAccount({ id: 'bank', code: '1010', name: 'Bank', type: 'asset', category: 'Cash' });
+const sales: ChartAccount = makeChartAccount({ id: 'sales', code: '4000', name: 'Sales', type: 'income', category: 'Revenue' });
+const rent: ChartAccount = makeChartAccount({ id: 'rent', code: '6100', name: 'Rent', type: 'expense', category: 'Operating Expenses' });
+const loan: ChartAccount = makeChartAccount({ id: 'loan', code: '2500', name: 'Bank loan', type: 'liability', category: 'Long-term Liabilities' });
+const accounts = [bank, sales, rent, loan];
+
+const entry = (id: string, date: string, lines: Array<[string, number]>) => makeTransaction({
+  id, date, description: `Entry ${id}`,
+  lines: lines.map(([accountId, amountCents]) => ({ id: `line-${id}-${accountId}`, accountId, amountCents })),
+});
+
+/** A sale: the bank up, sales up; balanced. */
+const sale = (id: string, date: string, cents: number) => entry(id, date, [['bank', cents], ['sales', -cents]]);
+const today = '2026-03-31';
+
+describe('Probing dates', () => {
+  test('only days that carry a postable entry, in order, then the end date', () => {
+    const transactions: Transaction[] = [
+      sale('c', '2026-03-20', 100),
+      sale('a', '2026-03-01', 100),
+      makeTransaction({ id: 'draft', date: '2026-03-05', status: 'draft', lines: [{ accountId: 'bank', amountCents: 1 }] }),
+      makeTransaction({ id: 'void', date: '2026-03-06', status: 'void', lines: [{ accountId: 'bank', amountCents: 1 }] }),
+      sale('late', '2026-04-02', 100),
+      makeTransaction({ id: 'stamp', date: '2026-03-10T12:00:00.000Z', lines: [{ accountId: 'bank', amountCents: 1 }, { accountId: 'sales', amountCents: -1 }] }),
+    ];
+    assert.deepEqual(probeDates(transactions, today), ['2026-03-01', '2026-03-10', '2026-03-20', today]);
+  });
+
+  test('the difference is the report code’s own two totals', () => {
+    assert.equal(balanceDifferenceCents(accounts, [sale('a', '2026-03-01', 1000)], today), 0);
+    assert.equal(balanceDifferenceCents(accounts, [entry('b', '2026-03-01', [['bank', 500], ['sales', -400]])], today), 100);
+    assert.equal(balanceDifferenceCents(accounts, [], today), 0);
+  });
+});

+describe('Finding the day the books stopped balancing', () => {
+  test('a break on the first day of the history', () => {
+    const transactions = [entry('first', '2026-03-01', [['bank', 500], ['sales', -400]])];
+    const result = findBalanceBreak(accounts, transactions, today);

+    assert.equal(result.firstUnbalancedDate, '2026-03-01');

+    assert.equal(result.differenceCents, 100);
